// TEMPORARY injection harness for the #766 Surface plate/fill/pipe batch (batch 13). This `.txt` is
// the evidence copy: for a sweep it is copied to Sources/OCCTSwift/SweepPlateFillShadow.swift and
// removed again, and that file must never be committed (README.md, "Running it"). Every function
// below has the name and C signature of an OCCTBridge entry point, so the module-local declaration

enum PF {

}

func OCCTShapePlatePoints(_ points: UnsafePointer<Double>, _ count: Int32, _ tol: Double) -> OCCTShapeRef? {
    if PF.on("PLATEPTS_DROP_LAST") { return OCCTBridge.OCCTShapePlatePoints(points, count - 1, tol) }
    if PF.on("PLATEPTS_Z_SHIFT") {
        var p = Array(UnsafeBufferPointer(start: points, count: Int(count) * 3))
        for i in 0..<Int(count) { p[i * 3 + 2] += 0.5 }
        return p.withUnsafeBufferPointer { OCCTBridge.OCCTShapePlatePoints($0.baseAddress!, count, tol) }
    }
    if PF.on("PLATEPTS_DROP_FIRST") { return OCCTBridge.OCCTShapePlatePoints(points + 3, count - 1, tol) }
    return OCCTBridge.OCCTShapePlatePoints(points, count, tol)
}

    _ base: OCCTShapeRef, _ profile: OCCTShapeRef, _ faceIndex: Int32, _ spine: OCCTWireRef, _ fuse: Int32
) -> OCCTShapeRef? {
    if PF.on("PF_FUSE_FLIPPED") { return OCCTBridge.OCCTShapePipeFeatureFromProfile(base, profile, faceIndex, spine, fuse == 0 ? 1 : 0) }
    return OCCTBridge.OCCTShapePipeFeatureFromProfile(base, profile, faceIndex, spine, fuse)
}
