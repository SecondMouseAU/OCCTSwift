// TEMPORARY injection harness for the #2983 Curve re-sweep. This `.txt` is the evidence copy: for
// a sweep it is copied to Sources/OCCTSwift/SweepCurveShadow.swift and removed again, and that
// file must never be committed (README.md, "Running it").
// One build serves every switch; SWEEP_CURVE_SWITCH selects one at runtime. With the variable
// unset every shadow is a pure pass-through, which the sweep asserts (suite green, no switch).

enum SweepCurveGate {
    static let active: String =
        Foundation.ProcessInfo.processInfo.environment["SWEEP_CURVE_SWITCH"] ?? ""
    static func on(_ name: String) -> Bool { active == name }
    static func hasPrefix(_ p: String) -> Bool { active.hasPrefix(p) }
}

private typealias G = SweepCurveGate

// MARK: - Ellipse constructors (centre, normal, radii) and the angle / point variants

/// The distortion every ellipse construction site shares. `nil` means the site must return nil.
private func ellipseRadii(_ a: Double, _ b: Double) -> (Double, Double)? {
    if G.on("ELL_NIL") { return nil }
    var (a, b) = (a, b)
    if G.on("ELL_SWAP_RADII") { (a, b) = (b, a) }
    if G.on("ELL_MINOR_EQ_MAJOR") { b = a }
    if G.on("ELL_MAJOR_EQ_MINOR") { a = b }
    if G.on("ELL_RADII_X2") { a *= 2; b *= 2 }
    if G.on("ELL_ZERO_ACCEPTED") && (a <= 0 || b <= 0) { a = max(a, 1); b = a }
    if G.on("ELL_NEGATIVE_ACCEPTED") { a = abs(a); b = abs(b) }
    if G.on("ELL_INVERTED_ACCEPTED") && b > a { (a, b) = (b, a) }
    return (a, b)
}

private func vec(_ x: Double, _ y: Double, _ z: Double, flip: Bool, drop: Bool)
    -> (Double, Double, Double)
{
    if drop { return (0, 0, 0) }
    return flip ? (-x, -y, -z) : (x, y, z)
}

func OCCTCurve3DArcOfEllipse(
    _ cx: Double, _ cy: Double, _ cz: Double, _ nx: Double, _ ny: Double, _ nz: Double,
    _ major: Double, _ minor: Double, _ a1: Double, _ a2: Double, _ sense: Bool
) -> OpaquePointer? {
    guard let (a, b) = ellipseRadii(major, minor) else { return nil }
    let c = vec(cx, cy, cz, flip: false, drop: G.on("ELL_CENTER_DROPPED"))
    let n = vec(nx, ny, nz, flip: G.on("ELL_NORMAL_FLIPPED"), drop: false)
    var (s, e) = (a1, a2)
    if G.on("ARC_ANGLE2_HALVED") { e = a2 / 2 }
    if G.on("ARC_ANGLE1_SHIFTED") { s = a1 + 0.1 }
    if G.on("ARC_ANGLES_SWAPPED") { (s, e) = (a2, a1) }
    let sense2 = G.on("ARC_SENSE_FLIPPED") ? !sense : sense
    return OCCTBridge.OCCTCurve3DArcOfEllipse(
        c.0, c.1, c.2, n.0, n.1, n.2, a, b, s, e, sense2)
}

// MARK: - Hyperbola

private func hyperbolaRadii(_ a: Double, _ b: Double) -> (Double, Double)? {
    if G.on("HYP_NIL") { return nil }
    var (a, b) = (a, b)
    if G.on("HYP_SWAP_RADII") { (a, b) = (b, a) }
    if G.on("HYP_RADII_X2") { a *= 2; b *= 2 }
    if G.on("HYP_ZERO_ACCEPTED") {
        if a <= 0 { a = 1 }
        if b <= 0 { b = 1 }
    }
    // The ellipse rule copied onto a hyperbola, which puts no ordering on its radii.
    if G.on("HYP_ORDER_ENFORCED") && b > a { return nil }
    return (a, b)
}

private func parabolaFocal(_ f: Double) -> Double? {
    if G.on("PAR_NIL") { return nil }

    if G.on("PAR_FOCAL_HALF") { f /= 2 }
    if G.on("PAR_ZERO_ACCEPTED") && f <= 0 { f = 1 }
    return f
}

// MARK: - The in-place setters and their getters

/// One setter site. `family` is the switch prefix (ELLM, ELLJ, HYPM, HYPJ, PARF, CIRR).
private func setter(
    _ family: String, _ r: Double, real: (Double) -> Bool, sibling: ((Double) -> Bool)? = nil,
    currentOther: Double? = nil
) -> Bool {
    if G.on("\(family)_NOOP_TRUE") { return true }
    if G.on("\(family)_FALSE") { return false }
    if G.on("\(family)_ZERO_ACCEPTED") && r == 0 { return real(1e-9) }
    if G.on("\(family)_WRONG_VALUE") { return real(r * 2) }
    if G.on("\(family)_SETS_SIBLING"), let sibling { return sibling(r) }
    if G.on("\(family)_BELOW_OTHER_ACCEPTED"), let o = currentOther, r > 0, r < o { return true }
    return real(r)
}

}

// MARK: - Extrema (point/ellipse, point/parabola, line/ellipse)

/// Distort the packed result array of one extrema entry point. `family` is XE, XP or XL.
private func distortExtrema(
    _ family: String, _ n: Int32, _ out: UnsafeMutablePointer<OCCTExtremaElResult>, _ max: Int32
) -> Int32 {
    var n = n
    if n <= 0 { return n }
    if G.on("\(family)_NIL") { return 0 }
    if G.on("\(family)_ONE") { n = 1 }
    if G.on("\(family)_DROP_LAST") { n -= 1 }
    if G.on("\(family)_DROP_FIRST") {
        for i in 0..<Int(n - 1) { out[i] = out[i + 1] }
        n -= 1
    }
    if G.on("\(family)_DUP_FIRST") && n < max {
        out[Int(n)] = out[0]
        n += 1
    }
    for i in 0..<Int(n) {
        if G.on("\(family)_SQDIST_X2") { out[i].squareDistance *= 2 }
        if G.on("\(family)_SQDIST_PLUS1") { out[i].squareDistance += 1 }
        if G.on("\(family)_SWAP_WITNESS") {
            let r = out[i]
            out[i].x1 = r.x2; out[i].y1 = r.y2; out[i].z1 = r.z2
            out[i].x2 = r.x1; out[i].y2 = r.y1; out[i].z2 = r.z1
        }
        if G.on("\(family)_POINT1_SHIFTED") { out[i].x1 += 1 }
        if G.on("\(family)_POINT2_SHIFTED") { out[i].x2 += 1 }
        if G.on("\(family)_PARALLEL_TRUE") { out[i].isParallel = true }
        if G.on("\(family)_NO_WITNESS") { out[i].hasWitnessPoints = false }
    }
    return n
}

func OCCTExtremaExtPElCElips(
    _ px: Double, _ py: Double, _ pz: Double, _ cx: Double, _ cy: Double, _ cz: Double,
    _ nx: Double, _ ny: Double, _ nz: Double, _ xdx: Double, _ xdy: Double, _ xdz: Double,
    _ major: Double, _ minor: Double, _ tol: Double,
    _ out: UnsafeMutablePointer<OCCTExtremaElResult>, _ max: Int32
) -> Int32 {
    var (a, b) = (major, minor)
    if G.on("XE_ZERO_ACCEPTED") && (a <= 0 || b <= 0) { a = Swift.max(a, 1); b = a }
    let n = OCCTBridge.OCCTExtremaExtPElCElips(
        px, py, pz, cx, cy, cz, nx, ny, nz, xdx, xdy, xdz, a, b, tol, out, max)
    return distortExtrema("XE", n, out, max)
}

// MARK: - The arc-length adaptors (WireCurve over CompCurve, EdgeCurve), prefixes W_ and E_

private func adaptorLength(_ p: String, _ real: Double) -> Double {
    if G.on("\(p)_LEN_PLUS") { return real + 1e-3 }
    if G.on("\(p)_LEN_HALF") { return real / 2 }
    if G.on("\(p)_LEN_NEG") { return -real }
    return real
}

private func adaptorRange(_ p: String, _ f: inout Double, _ l: inout Double) {
    if G.on("\(p)_PR_SWAP") { swap(&f, &l) }
    if G.on("\(p)_PR_FIRST_SHIFT") { f += 0.25 }
    if G.on("\(p)_PR_LAST_SHIFT") { l -= 0.25 }
}

private func adaptorPoint(
    _ p: String, _ u: Double, _ x: UnsafeMutablePointer<Double>, _ y: UnsafeMutablePointer<Double>,
    _ z: UnsafeMutablePointer<Double>, real: (Double) -> Bool
) -> Bool {
    if G.on("\(p)_PT_FALSE") { return false }
    let ok = G.on("\(p)_PT_PARAM_SHIFT") ? real(u + 0.05) : real(u)
    if ok && G.on("\(p)_PT_XY_SWAPPED") { swap(&x.pointee, &y.pointee) }
    if ok && G.on("\(p)_PT_Z_SHIFT") { z.pointee += 1e-3 }
    return ok
}

private func adaptorTangent(
    _ p: String, _ x: UnsafeMutablePointer<Double>, _ y: UnsafeMutablePointer<Double>,
    _ z: UnsafeMutablePointer<Double>, real: () -> Bool
) -> Bool {
    if G.on("\(p)_TG_FALSE") { return false }
    let ok = real()
    if ok && G.on("\(p)_TG_NEGATED") { x.pointee = -x.pointee; y.pointee = -y.pointee; z.pointee = -z.pointee }
    if ok && G.on("\(p)_TG_X2") { x.pointee *= 2; y.pointee *= 2; z.pointee *= 2 }
    if ok && G.on("\(p)_TG_XY_SWAPPED") { swap(&x.pointee, &y.pointee) }
    return ok
}

private func adaptorAbscissa(
    _ p: String, _ s: Double, _ out: UnsafeMutablePointer<Double>, range: (Double, Double),
    real: (Double) -> Bool
) -> Bool {
    if G.on("\(p)_PA_FALSE") { return false }
    if G.on("\(p)_PA_IDENTITY") { out.pointee = s; return true }
    if G.on("\(p)_PA_REVERSED") {
        let ok = real(s)
        if ok { out.pointee = range.1 - (out.pointee - range.0) }
        return ok
    }
    let ok = real(s)
    if ok && G.on("\(p)_PA_SHIFT") { out.pointee += 0.05 }
    return ok
}

private func adaptorSample(
    _ p: String, _ count: Int32, _ out: UnsafeMutablePointer<Double>, real: () -> Int32
) -> Int32 {
    var n = real()
    if G.on("\(p)_SU_WRITE_NONE") { return 0 }
    if G.on("\(p)_SU_DROP_LAST") { n -= 1 }
    let c = Int(n)
    if c >= 2 {
        if G.on("\(p)_SU_REVERSED") {
            var i = 0
            var j = c - 1
            while i < j {
                for k in 0..<3 { swap(&out[i * 3 + k], &out[j * 3 + k]) }
                i += 1; j -= 1
            }
        }
        if G.on("\(p)_SU_DUP_FIRST") { for k in 0..<3 { out[3 + k] = out[k] } }
        if G.on("\(p)_SU_LAST_IS_FIRST") { for k in 0..<3 { out[(c - 1) * 3 + k] = out[k] } }
        if G.on("\(p)_SU_FIRST_MOVED") { out[0] += 0.01 }
        if G.on("\(p)_SU_MIDPOINTS") {
            // each sample replaced by the midpoint of itself and its successor: evenly spaced
            // along the wire but at the wrong abscissae
            for i in 0..<(c - 1) { for k in 0..<3 { out[i * 3 + k] = (out[i * 3 + k] + out[(i + 1) * 3 + k]) / 2 } }
        }
    }
    return n
}

// MARK: - Law knot splitting (bridge side; the Swift retry loop is edited by inject.py)

func OCCTLawBSplineKnotSplitting(
    _ law: OpaquePointer, _ order: Int32, _ out: UnsafeMutablePointer<Int32>, _ maxIdx: Int32
) -> Int32 {
    var n = OCCTBridge.OCCTLawBSplineKnotSplitting(law, order, out, maxIdx)
    if n < 0 && G.on("LAW_IDX_READS_UNREADABLE") && maxIdx >= 2 {
        out[0] = 1; out[1] = 2
        return 2
    }
    if n < 0 { return n }
    if G.on("LAW_IDX_NIL") { return -1 }
    let w = Int(Swift.min(n, maxIdx))
    if G.on("LAW_IDX_CAP100") { n = Swift.min(n, 100) }
    if G.on("LAW_IDX_COUNT_PLUS1") { n += 1 }
    if G.on("LAW_IDX_COUNT_MINUS1") { n -= 1 }
    for i in 0..<w {
        if G.on("LAW_IDX_SHIFT_UP") { out[i] += 1 }
        if G.on("LAW_IDX_ZERO_BASED") { out[i] -= 1 }
    }
    if G.on("LAW_IDX_REVERSED") && w > 1 {
        var i = 0, j = w - 1
        while i < j { swap(&out[i], &out[j]); i += 1; j -= 1 }
    }
    if G.on("LAW_IDX_FIRST_REPEATED") && w > 1 { out[1] = out[0] }
    return n
}

func OCCTLawBSplineKnotSplitParams(
    _ law: OpaquePointer, _ order: Int32, _ out: UnsafeMutablePointer<Double>, _ maxP: Int32
) -> Int32 {
    var n = OCCTBridge.OCCTLawBSplineKnotSplitParams(law, order, out, maxP)
    if n < 0 && G.on("LAW_PAR_READS_UNREADABLE") && maxP >= 2 {
        out[0] = 0; out[1] = 1
        return 2
    }
    if n < 0 { return n }
    if G.on("LAW_PAR_NIL") { return -1 }
    let w = Int(Swift.min(n, maxP))
    if G.on("LAW_PAR_CAP100") { n = Swift.min(n, 100) }
    if G.on("LAW_PAR_COUNT_PLUS1") { n += 1 }
    if G.on("LAW_PAR_COUNT_MINUS1") { n -= 1 }
    for i in 0..<w {
        if G.on("LAW_PAR_SHIFT") { out[i] += 0.5 }
        if G.on("LAW_PAR_AS_INDEX") { out[i] = Double(i + 1) * 1.0000001 }
    }
    if G.on("LAW_PAR_REVERSED") && w > 1 {
        var i = 0, j = w - 1
        while i < j { swap(&out[i], &out[j]); i += 1; j -= 1 }
    }
    if G.on("LAW_PAR_FIRST_REPEATED") && w > 1 { out[1] = out[0] }
    return n
}

// MARK: - The two three-point conic forms (Issue554's deliberately unguarded sites)

func OCCTGCMakeEllipse3Points(
    _ x1: Double, _ y1: Double, _ z1: Double, _ x2: Double, _ y2: Double, _ z2: Double,
    _ cx: Double, _ cy: Double, _ cz: Double
) -> OpaquePointer? {
    if G.on("GC3E_NIL") { return nil }
    if G.on("GC3E_DEGENERATE_ACCEPTED") {
        let degenerate = (x1 == x2 && y1 == y2 && z1 == z2) || (y1 == 0 && z1 == 0 && y2 == 0 && z2 == 0)
        if degenerate {
            return OCCTBridge.OCCTGCMakeEllipse3Points(5, 0, 0, 0, 3, 0, cx, cy, cz)
        }
    }
    if G.on("GC3E_SWAP_S1_S2") {
        return OCCTBridge.OCCTGCMakeEllipse3Points(x2, y2, z2, x1, y1, z1, cx, cy, cz)
    }
    if G.on("GC3E_CENTER_DROPPED") {
        return OCCTBridge.OCCTGCMakeEllipse3Points(x1, y1, z1, x2, y2, z2, 0, 0, 0)
    }
    return OCCTBridge.OCCTGCMakeEllipse3Points(x1, y1, z1, x2, y2, z2, cx, cy, cz)
}

func OCCTGCMakeHyperbola3Points(
    _ x1: Double, _ y1: Double, _ z1: Double, _ x2: Double, _ y2: Double, _ z2: Double,
    _ cx: Double, _ cy: Double, _ cz: Double
) -> OpaquePointer? {
    if G.on("GC3H_NIL") { return nil }
    let r = OCCTBridge.OCCTGCMakeHyperbola3Points(x1, y1, z1, x2, y2, z2, cx, cy, cz)
    if r == nil && G.on("GC3H_DEGENERATE_ACCEPTED") {
        return OCCTBridge.OCCTGCMakeHyperbola3Points(5, 0, 0, 5, 3, 0, 0, 0, 0)
    }
    return r
}

// MARK: - BSpline reads (BSplineMutationsTests)

func OCCTCurve3DBSplineGetKnotSequence(
    _ curve: OpaquePointer, _ seq: UnsafeMutablePointer<Double>, _ maxCount: Int32
) -> Int32 {
    var n = OCCTBridge.OCCTCurve3DBSplineGetKnotSequence(curve, seq, maxCount)
    if G.on("BS_SEQ_EMPTY") { return 0 }
    if G.on("BS_SEQ_DROP_LAST") { n -= 1 }
    let c = Int(n)
    if G.on("BS_SEQ_UNEXPANDED") && c > 0 {
        // distinct knots only: the multiplicities are lost
        var out = [Double]()
        for i in 0..<c where out.last != seq[i] { out.append(seq[i]) }
        for (i, v) in out.enumerated() { seq[i] = v }
        return Int32(out.count)
    }
    if G.on("BS_SEQ_REVERSED") && c > 1 {
        var i = 0, j = c - 1
        while i < j { swap(&seq[i], &seq[j]); i += 1; j -= 1 }
    }
    if G.on("BS_SEQ_SHIFT") { for i in 0..<c { seq[i] += 0.5 } }
    if G.on("BS_SEQ_ZEROED") { for i in 0..<c { seq[i] = 0 } }
    return n
}

func OCCTCurve3DBSplineGetWeights(_ curve: OpaquePointer, _ w: UnsafeMutablePointer<Double>) {

    if G.on("BS_W_X2") { for i in 0..<n { w[i] *= 2 } }
    if G.on("BS_W_LAST_WRONG") && n > 0 { w[n - 1] = 2 }
}

    let c = OCCTBridge.OCCTCurve3DGetContinuity(curve)
    if G.on("CONT_PLUS1") { return c + 1 }
    if G.on("CONT_C0") { return 0 }
    if G.on("CONT_CN_AS_99") && c == 6 { return 99 }
    if G.on("CONT_G1_AS_NEG2") && c == 1 { return -2 }

func OCCTEdgeProjectPoint(_ edge: OpaquePointer, _ px: Double, _ py: Double, _ pz: Double)
    -> OCCTCurveProjectionResult
{
    var r = OCCTBridge.OCCTEdgeProjectPoint(edge, px, py, pz)
    if G.on("PRJ_INVALID") { r.isValid = false }
    if G.on("PRJ_DIST_X2") { r.distance *= 2 }
    if G.on("PRJ_DIST_PLUS1") { r.distance += 1 }
    if G.on("PRJ_DIST_ZERO") { r.distance = 0 }
    if G.on("PRJ_POINT_SHIFT") { r.px += 1 }
    if G.on("PRJ_POINT_IS_QUERY") { r.px = px; r.py = py; r.pz = pz }
    if G.on("PRJ_PARAM_SHIFT") { r.parameter += 0.5 }
    return r
}

func OCCTCurve3DProjectPoint(_ curve: OpaquePointer, _ px: Double, _ py: Double, _ pz: Double, _ precision: Double)
    -> OCCTCurveProjectResult
{
    var r = OCCTBridge.OCCTCurve3DProjectPoint(curve, px, py, pz, precision)
    if G.on("PRJ_DIST_X2") { r.distance *= 2 }
    if G.on("PRJ_DIST_PLUS1") { r.distance += 1 }
    if G.on("PRJ_DIST_ZERO") { r.distance = 0 }
    if G.on("PRJ_POINT_SHIFT") { r.projX += 1 }
    if G.on("PRJ_POINT_IS_QUERY") { r.projX = px; r.projY = py; r.projZ = pz }
    if G.on("PRJ_PARAM_SHIFT") { r.parameter += 0.5 }
    return r
}

func OCCTPointsToBSplineWithParams(
    _ pts: UnsafePointer<Double>, _ count: Int32, _ degMin: Int32, _ degMax: Int32,
    _ continuity: Int32, _ tol: Double
) -> OpaquePointer? {
    if G.on("P2B_NIL") { return nil }
    if G.on("P2B_FAILS_ABOVE_C2") && continuity > 2 { return nil }
    if G.on("P2B_DROP_LAST_POINT") && count > 2 {
        return OCCTBridge.OCCTPointsToBSplineWithParams(pts, count - 1, degMin, degMax, continuity, tol)
    }
    if G.on("P2B_REVERSED") {
        var r = [Double]()
        for i in stride(from: Int(count) - 1, through: 0, by: -1) {
            r += [pts[i * 3], pts[i * 3 + 1], pts[i * 3 + 2]]
        }
        return r.withUnsafeBufferPointer {
            OCCTBridge.OCCTPointsToBSplineWithParams($0.baseAddress!, count, degMin, degMax, continuity, tol)
        }
    }
    if G.on("P2B_SHIFTED") {
        var r = [Double]()
        for i in 0..<Int(count) { r += [pts[i * 3] + 0.1, pts[i * 3 + 1], pts[i * 3 + 2]] }
        return r.withUnsafeBufferPointer {
            OCCTBridge.OCCTPointsToBSplineWithParams($0.baseAddress!, count, degMin, degMax, continuity, tol)
        }
    }
    return OCCTBridge.OCCTPointsToBSplineWithParams(pts, count, degMin, degMax, continuity, tol)
}
