    if GF.on("GAC_TN_SWAP") { swap3(tX, tY, tZ, nX, nY, nZ) }
    if GF.on("GAC_NB_SWAP") { swap3(nX, nY, nZ, bX, bY, bZ) }

private func swap3(
    _ aX: UnsafeMutablePointer<Double>, _ aY: UnsafeMutablePointer<Double>, _ aZ: UnsafeMutablePointer<Double>,

func OCCTGeomFillLocationDraftD0(
    _ ref: OCCTLocationDraftRef, _ param: Double, _ mat: UnsafeMutablePointer<Double>,
    _ vecX: UnsafeMutablePointer<Double>, _ vecY: UnsafeMutablePointer<Double>, _ vecZ: UnsafeMutablePointer<Double>
) -> Bool {
    if GF.on("LD_D0_FALSE") { return false }
    let ok = OCCTBridge.OCCTGeomFillLocationDraftD0(ref, param, mat, vecX, vecY, vecZ)
    if GF.on("LD_MAT_TRANSPOSE") {
        var t = Array(UnsafeBufferPointer(start: mat, count: 9))
        for r in 0..<3 { for c in 0..<3 { t[r * 3 + c] = mat[c * 3 + r] } }
        for i in 0..<9 { mat[i] = t[i] }
    }
    if GF.on("LD_MAT_ROW1_NEG") { for i in 3..<6 { mat[i] = -mat[i] } }
    if GF.on("LD_MAT_IDENTITY") { for i in 0..<9 { mat[i] = (i % 4 == 0) ? 1 : 0 } }
    if GF.on("LD_TRANS_X_PLUS") { vecX.pointee += 1e-6 }
    if GF.on("LD_TRANS_ZERO") { vecX.pointee = 0; vecY.pointee = 0; vecZ.pointee = 0 }
    return ok
}

// MARK: - GeomFill_Stretch, GeomFill_Sweep, Frenet and ConstantBiNormal trihedra

func OCCTGeomFillStretch(
    _ p1: UnsafePointer<Double>, _ p2: UnsafePointer<Double>, _ p3: UnsafePointer<Double>,
    _ p4: UnsafePointer<Double>, _ count: Int32, _ outPoles: UnsafeMutablePointer<Double>?, _ maxPoles: Int32
) -> OCCTStretchFillResult {
    if GF.on("STRETCH_ROWS_P1_P3") {
        return OCCTBridge.OCCTGeomFillStretch(p3, p2, p1, p4, count, outPoles, maxPoles)
    }
    var r = OCCTBridge.OCCTGeomFillStretch(p1, p2, p3, p4, count, outPoles, maxPoles)
    if let o = outPoles {
        let n = Int(r.nbUPoles) * Int(r.nbVPoles)
        if GF.on("STRETCH_CENTRE_Z_PLUS") && n == 9 { o[4 * 3 + 2] += 0.5 }
        if GF.on("STRETCH_POLES_1_3_SWAP") && n == 9 {
            for k in 0..<3 { let t = o[1 * 3 + k]; o[1 * 3 + k] = o[3 * 3 + k]; o[3 * 3 + k] = t }
        }
    }
    if GF.on("STRETCH_NU_PLUS1") { r.nbUPoles += 1 }
    if GF.on("STRETCH_RATIONAL") { r.isRational = true }
    return r
}

    if GF.on("GFS_NIL") { return nil }

    guard let r = OCCTBridge.OCCTGeomFillSweep(path, section) else { return nil }

func OCCTGeomFillFrenetTrihedron(_ edge: OCCTShapeRef, _ param: Double) -> OCCTTrihedronFrame {
    var f = OCCTBridge.OCCTGeomFillFrenetTrihedron(edge, GF.on("FR_PARAM_PLUS") ? param + 0.5 : param)
    if GF.on("FR_TN_SWAP") { swapFrame(&f.tx, &f.ty, &f.tz, &f.nx, &f.ny, &f.nz) }
    if GF.on("FR_NB_SWAP") { swapFrame(&f.nx, &f.ny, &f.nz, &f.bx, &f.by, &f.bz) }
    if GF.on("FR_N_NEG") { f.nx = -f.nx; f.ny = -f.ny; f.nz = -f.nz }
    if GF.on("FR_B_NEG") { f.bx = -f.bx; f.by = -f.by; f.bz = -f.bz }
    return f
}

func OCCTGeomFillConstantBiNormalTrihedron(
    _ edge: OCCTShapeRef, _ param: Double, _ bx: Double, _ by: Double, _ bz: Double
) -> OCCTTrihedronFrame {
    var f: OCCTTrihedronFrame
    if GF.on("CB_BINORMAL_NEG") {
        f = OCCTBridge.OCCTGeomFillConstantBiNormalTrihedron(edge, param, -bx, -by, -bz)
    } else if GF.on("CB_BINORMAL_X") {
        f = OCCTBridge.OCCTGeomFillConstantBiNormalTrihedron(edge, param, 1, 0, 0)
    } else {
        f = OCCTBridge.OCCTGeomFillConstantBiNormalTrihedron(edge, param, bx, by, bz)
    }
    if GF.on("CB_TN_SWAP") { swapFrame(&f.tx, &f.ty, &f.tz, &f.nx, &f.ny, &f.nz) }
    if GF.on("CB_N_NEG") { f.nx = -f.nx; f.ny = -f.ny; f.nz = -f.nz }
    return f
}

private func swapFrame(
    _ aX: inout Double, _ aY: inout Double, _ aZ: inout Double,
    _ bX: inout Double, _ bY: inout Double, _ bZ: inout Double
) {
    let x = aX, y = aY, z = aZ
    aX = bX; aY = bY; aZ = bZ
    bX = x; bY = y; bZ = z
}

func OCCTPlateLoadLinearXYZ(
    _ plate: OCCTPlateRef, _ uvs: UnsafePointer<Double>, _ targets: UnsafePointer<Double>,
    _ coeffs: UnsafePointer<Double>, _ count: Int32
) -> Bool {
    if GF.on("PLATE_LXYZ_NOT_LOADED") { return true }
    if GF.on("PLATE_LXYZ_COEFF_SECOND_PLUS") {
        // F(0,0) - F(1,0) = 0 becomes F(0,0) + F(1,0) = 0: (1, 0) is pulled to -z instead of +z.
        var c = (0..<Int(count)).map { coeffs[$0] }
        if c.count > 1 { c[1] = -c[1] }
        return c.withUnsafeBufferPointer {
            OCCTBridge.OCCTPlateLoadLinearXYZ(plate, uvs, targets, $0.baseAddress!, count)
        }
    }
    if GF.on("PLATE_LXYZ_COEFF_SECOND_ZERO") {
        var c = (0..<Int(count)).map { coeffs[$0] }
        if c.count > 1 { c[1] = 0 }
        return c.withUnsafeBufferPointer {
            OCCTBridge.OCCTPlateLoadLinearXYZ(plate, uvs, targets, $0.baseAddress!, count)
        }
    }
    return OCCTBridge.OCCTPlateLoadLinearXYZ(plate, uvs, targets, coeffs, count)
}

// MARK: - GeomPlate_BuildAveragePlane

func OCCTGeomPlateBuildAveragePlane(
    _ points: UnsafePointer<Double>, _ pointCount: Int32, _ nbBound: Int32, _ tol: Double
) -> OCCTAveragePlaneResult {
    var r = OCCTBridge.OCCTGeomPlateBuildAveragePlane(points, pointCount, nbBound, tol)
    if GF.on("AVG_NORMAL_NEG") { r.normalX = -r.normalX; r.normalY = -r.normalY; r.normalZ = -r.normalZ }
    if GF.on("AVG_NORMAL_XZ_SWAP") { let t = r.normalX; r.normalX = r.normalZ; r.normalZ = t }
    if GF.on("AVG_ORIGIN_X_PLUS") { r.originX += 1e-6 }
    if GF.on("AVG_UV_U_SWAP") { let t = r.umin; r.umin = r.umax; r.umax = t }
    if GF.on("AVG_UV_V_NARROW") { r.vmax -= 1e-6 }
    if GF.on("AVG_IS_LINE") { r.isLine = true; r.isPlane = false }
    if GF.on("AVG_NOT_PLANE") { r.isPlane = false }
    return r
}
