// TEMPORARY injection harness for the #766 Surface batch 13 (revolution, BSpline/Bezier
// completion, conversion and projection tests). This `.txt` is the evidence copy: for a sweep it is
// copied to Sources/OCCTSwift/SweepSurfaceShadow.swift and removed again, and that file must never
// be committed (README.md, "Running it"). Every function below has the same name and C signature as
// an OCCTBridge entry point, so the module-local declaration wins over the imported one at every
// call site with no edit to any call site or any .mm. One build serves every switch; SF_SWITCH
// selects one at runtime. With the variable unset every shadow is a pure pass-through, which the
// sweep asserts (suites green, no switch).

enum SF {
    static let active: String = Foundation.ProcessInfo.processInfo.environment["SF_SWITCH"] ?? ""
    static func on(_ name: String) -> Bool { active == name }

    /// A heap copy of a double array (leaked on purpose, the harness is temporary).
    static func copy(_ p: UnsafePointer<Double>, _ n: Int, _ f: (Int, Double) -> Double)
        -> UnsafePointer<Double>
    {
        let q = UnsafeMutablePointer<Double>.allocate(capacity: max(n, 1))
        for i in 0..<n { q[i] = f(i, p[i]) }
        return UnsafePointer(q)
    }
}

func OCCTGeomEvalHyperboloidCreate(_ r1: Double, _ r2: Double, _ mode: Int32) -> OCCTSurfaceRef? {
    if SF.on("HYPERCREATE_MODE_FLIPPED") {
        return OCCTBridge.OCCTGeomEvalHyperboloidCreate(r1, r2, 1 - mode)
    }
    if SF.on("HYPERCREATE_R1_X1_1") { return OCCTBridge.OCCTGeomEvalHyperboloidCreate(r1 * 1.1, r2, mode) }
    return OCCTBridge.OCCTGeomEvalHyperboloidCreate(r1, r2, mode)
}

    }
    return OCCTBridge.OCCTGeomEvalHyperboloidD0(r1, r2, mode, u, v, px, py, pz)

func OCCTGeomEvalCircularHelixCurveCreate(_ radius: Double, _ pitch: Double) -> OCCTCurve3DRef? {
    if SF.on("CHELIX_PITCH_X2") { return OCCTBridge.OCCTGeomEvalCircularHelixCurveCreate(radius, pitch * 2) }
    return OCCTBridge.OCCTGeomEvalCircularHelixCurveCreate(radius, pitch)
}

func OCCTExtremaPCMinDistance(_ c: OCCTCurve3DRef, _ x: Double, _ y: Double, _ z: Double) -> Double {
    let d = OCCTBridge.OCCTExtremaPCMinDistance(c, x, y, z)
    if SF.on("EXTREMAPC_SQUARED") { return d * d }
    if SF.on("EXTREMAPC_PLUS_001") { return d + 0.001 }
    return d
}

func OCCTGeomEvalAHTBezierCurveCreate(
    _ poles: UnsafePointer<Double>, _ n: Int32, _ alg: Int32, _ alpha: Double, _ beta: Double
) -> OCCTCurve3DRef? {
    if SF.on("AHT_ALPHA_X2") { return OCCTBridge.OCCTGeomEvalAHTBezierCurveCreate(poles, n, alg, alpha * 2, beta) }
    if SF.on("AHT_ALGDEGREE_PLUS1") { return OCCTBridge.OCCTGeomEvalAHTBezierCurveCreate(poles, n, alg + 1, alpha, beta) }
    return OCCTBridge.OCCTGeomEvalAHTBezierCurveCreate(poles, n, alg, alpha, beta)
}

func OCCTGeomEvalAHTBezierCurveCreateRational(

    _ alpha: Double, _ beta: Double

        return OCCTBridge.OCCTGeomEvalAHTBezierCurveCreateRational(
            poles, SF.copy(weights, Int(n)) { _, _ in 1 }, n, alg, alpha, beta)
    }
    return OCCTBridge.OCCTGeomEvalAHTBezierCurveCreateRational(poles, weights, n, alg, alpha, beta)
}
