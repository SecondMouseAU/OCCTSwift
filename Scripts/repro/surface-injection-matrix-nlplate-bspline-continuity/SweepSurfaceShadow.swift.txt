// TEMPORARY injection harness for the #766 Surface batch (batch 12). This `.txt` is the evidence
// copy: for a sweep it is copied to Sources/OCCTSwift/SweepSurfaceShadow.swift and removed again,
// and that file must never be committed (README.md, "Running it"). Every function below has the
// same name and C signature as an OCCTBridge entry point, so the module-local declaration wins
// over the imported one at every call site with no edit to any call site or any .mm.
// One build serves every switch; SF_SWITCH selects one at runtime. With the variable unset every
// shadow is a pure pass-through, which the sweep asserts (suites green, no switch).

enum SF {
    static let active: String = Foundation.ProcessInfo.processInfo.environment["SF_SWITCH"] ?? ""
    static func on(_ name: String) -> Bool { active == name }

    /// A stable C string for the typeName switches (leaked on purpose, the harness is temporary).
    static func cstr(_ s: String) -> UnsafePointer<CChar>? { UnsafePointer(strdup(s)) }
}

// MARK: - NLPlate and plate (NLPlateDeformation, NLPlateG2G3, ParametricPlateSurface)

private func SFCopy(_ c: UnsafePointer<Double>?, _ n: Int32, stride: Int) -> [Double] {
    guard let c else { return [] }
    return Array(UnsafeBufferPointer(start: c, count: Int(n) * stride))
}

/// The (u, v, x, y, z) head of every constraint, which is all a G0 solve reads.
private func SFG0Head(_ c: UnsafePointer<Double>?, _ n: Int32, stride: Int) -> [Double] {
    let all = SFCopy(c, n, stride: stride)
    var out: [Double] = []
    for i in 0..<Int(n) { out += all[(i * stride)..<(i * stride + 5)] }
    return out
}

func OCCTSurfaceNLPlateG0(
    _ s: OCCTSurfaceRef?, _ c: UnsafePointer<Double>?, _ n: Int32, _ order: Int32, _ tol: Double
) -> OCCTSurfaceRef? {
    var arr = SFCopy(c, n, stride: 5)
    if SF.on("NL_G0_Z_PLUS") { for i in 0..<Int(n) { arr[i * 5 + 4] += 0.5 } }
    if SF.on("NL_G0_UV_U_PLUS") { for i in 0..<Int(n) { arr[i * 5] += 0.5 } }
    return OCCTBridge.OCCTSurfaceNLPlateG0(s, arr, n, order, tol)
}

func OCCTSurfaceNLPlateG1(
    _ s: OCCTSurfaceRef?, _ c: UnsafePointer<Double>?, _ n: Int32, _ order: Int32, _ tol: Double
) -> OCCTSurfaceRef? {
    // "The fix": hand the position head to the G0 solve, which meets a single target. A known
    // issue pinned against the real G1 answer must go red when the answer starts being right.
    if SF.on("FIX_G1_AS_G0") {
        return OCCTBridge.OCCTSurfaceNLPlateG0(s, SFG0Head(c, n, stride: 11), n, order, tol)
    }
    return OCCTBridge.OCCTSurfaceNLPlateG1(s, c, n, order, tol)
}

func OCCTGeomEvalTBezierSurfaceCreate(
    _ poles: UnsafePointer<Double>, _ uCount: Int32, _ vCount: Int32, _ alphaU: Double, _ alphaV: Double
) -> OCCTSurfaceRef? {
    if SF.on("TBEZIER_SURFACE_TRANSPOSED") {
        var t = [Double](repeating: 0, count: Int(uCount * vCount) * 3)
        for i in 0..<Int(uCount) {
            for j in 0..<Int(vCount) {
                for k in 0..<3 { t[(j * Int(uCount) + i) * 3 + k] = poles[(i * Int(vCount) + j) * 3 + k] }
            }
        }
        return OCCTBridge.OCCTGeomEvalTBezierSurfaceCreate(t, vCount, uCount, alphaV, alphaU)
    }
    return OCCTBridge.OCCTGeomEvalTBezierSurfaceCreate(poles, uCount, vCount, alphaU, alphaV)
}

    _ surfMaxDX: Double, _ surfMaxDY: Double, _ surfMaxDZ: Double, _ surfMinDX: Double,
    _ surfMinDY: Double, _ surfMinDZ: Double, _ surfMaxCurv: Double, _ surfMinCurv: Double,
    _ tolerance: Double, _ surfOrientation: Int32, _ boundOrientation: Int32,
    _ outStateBefore: UnsafeMutablePointer<Int32>, _ outStateAfter: UnsafeMutablePointer<Int32>
) {
    OCCTBridge.OCCTTopTransSurfaceTransitionCurvature(
        tgtX, tgtY, tgtZ, normX, normY, normZ, maxDX, maxDY, maxDZ, minDX, minDY, minDZ, maxCurv, minCurv,
        surfNormX, surfNormY, surfNormZ, surfMaxDX, surfMaxDY, surfMaxDZ, surfMinDX, surfMinDY, surfMinDZ,
        surfMaxCurv, surfMinCurv, tolerance, surfOrientation, boundOrientation, outStateBefore, outStateAfter)
    if SF.on("TOPTRANS_STATES_SWAPPED") {
        let t = outStateBefore.pointee
        outStateBefore.pointee = outStateAfter.pointee
        outStateAfter.pointee = t
    }
}
